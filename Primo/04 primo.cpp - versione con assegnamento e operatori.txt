#include <assert.h>
#include <stdlib.h>
#include <stdio.h>
#include <stdbool.h>

int cmpint(const void *a, const void *b)
{
    const int *ia = (const int *)a;
    const int *ib = (const int *)b;
    if (*ia > *ib) {
        return 1;
    }
    if (*ia < *ib) {
        return -1;
    }
    return 0;
}

struct vector {
    int *data_;
    size_t n_;

    vector() {
        data_ = NULL;
        n_ = 0;
    }
    vector(size_t dim) {
        n_ = dim;
        data_ = new int[n_];
    }
    vector(const vector &other) {
        n_ = other.n_;
        data_ = new int[n_];
        for (size_t i = 0; i < n_; ++i) {
            data_[i] = other.data_[i];
        }
    }
    vector& operator=(const vector &rhs) {
        if (n_ < rhs.n_) {
            delete[] data_;
            data_ = new int[rhs.n_];
        }
        n_ = rhs.n_;
        for (size_t i = 0; i < n_; ++i) {
            data_[i] = rhs.data_[i];
        }
        return *this;
    }
    ~vector() {
        delete[] data_;
    }
    void pushback(int val) {
        // Alloco lo spazio per la nuova memoria
        int *newmem = new int[n_ + 1];
        // Copio lì dentro i dati vecchi
        for (size_t i = 0; i < n_; ++i) {
            newmem[i] = data_[i];
        }
        // Libero la vecchia memoria
        delete[] data_;
        n_++;
        data_ = newmem;
        data_[n_ - 1] = val;
    }
    void sort() {
        qsort(data_, n_, sizeof(int), cmpint);
    }
    size_t size() const {
        return n_;
    }
    int at(size_t pos) const {
        assert(pos < n_);
        return data_[pos];
    }
    int& operator[](size_t pos) {
        return data_[pos];
    }
    const int& operator[](size_t pos) const {
        return data_[pos];
    }
};

int main(int argc, char *argv[])
{    
    // Controllo il numero di argomenti
    if (argc != 3) {
        return 1;
    }
    // Creo un vettore vuoto
    vector v;
    // Apro il file
    FILE *fin = fopen(argv[1], "r");
    if (fin == NULL) {
        return 1;
    }
    // Leggo il contenuto 
    int val;
    while (fscanf(fin, "%i", &val) == 1) {
        // Ho un valore, allora lo aggiungo in coda al vettore
        v.pushback(val);
    }
    // Chiudo il file
    fclose(fin);

    // Ordino il vettore
    v.sort();

    // Creo il nuovo file
    FILE *fout = fopen(argv[2], "w");
    if (fout == NULL) {
        return 1;
    }
    // Scrivo i valori su file
    for (size_t i = 0; i < v.size(); ++i) {
        fprintf(fout, "%i\n", v[i]);
    }
    // Chiudo il file
    fclose(fout);

    return 0;
}
