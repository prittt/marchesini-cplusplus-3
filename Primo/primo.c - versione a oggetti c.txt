#include <assert.h>
#include <stdlib.h>
#include <stdio.h>
#include <stdbool.h>

int cmpint(const void *a, const void *b)
{
    const int *ia = a;
    const int *ib = b;
    if (*ia > *ib) {
        return 1;
    }
    if (*ia < *ib) {
        return -1;
    }
    return 0;
}

typedef struct vector {
    int *data;
    size_t n;
} vector;
void vector_constructor(vector *this) {
    this->data = NULL;
    this->n = 0;
}
void vector_destructor(vector *this) {
    free(this->data);
}
vector *new_vector() {
    vector *v = malloc(sizeof v);
    vector_constructor(v);
    return v;
}
void delete_vector(vector *this) {
    vector_destructor(this);
    free(this);
}
void vector_pushback(vector *this, int val) {
    this->n++;
    this->data = realloc(this->data, this->n * sizeof(int));
    this->data[this->n - 1] = val;
}
void vector_sort(vector *this) {
    qsort(this->data, this->n, sizeof(int), cmpint);
}
size_t vector_size(const vector *this) {
    return this->n;
}
int vector_at(const vector *this, size_t pos) {
    assert(pos < this->n);
    return this->data[pos];
}

int main(int argc, char *argv[])
{    
    // Controllo il numero di argomenti
    if (argc != 3) {
        return 1;
    }
    // Creo un vettore vuoto
    vector v;
    vector_constructor(&v);
    // Apro il file
    FILE *fin = fopen(argv[1], "r");
    if (fin == NULL) {
        return 1;
    }
    // Leggo il contenuto 
    int val;
    while (fscanf(fin, "%i", &val) == 1) {
        // Ho un valore, allora lo aggiungo in coda al vettore
        vector_pushback(&v, val);
    }
    // Chiudo il file
    fclose(fin);

    // Ordino il vettore
    vector_sort(&v);

    // Creo il nuovo file
    FILE *fout = fopen(argv[2], "w");
    if (fout == NULL) {
        return 1;
    }
    // Scrivo i valori su file
    for (size_t i = 0; i < vector_size(&v); ++i) {
        fprintf(fout, "%i\n", vector_at(&v, i));
    }
    // Chiudo il file
    fclose(fout);
    // Libero la memoria
    vector_destructor(&v);
    return 0;
}
